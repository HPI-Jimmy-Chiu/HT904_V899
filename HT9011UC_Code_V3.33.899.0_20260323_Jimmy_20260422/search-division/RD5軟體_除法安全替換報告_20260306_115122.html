安全除法模板定義

// MachineType.h
template <class T>
float ChangeToFloatNonPcnt(const T Numerator, const T Denominator)
{
    double str=0.00;
    if(Denominator!=0)
       str= ((double)Numerator/(double)Denominator);
    return str;
};
